// Adaptive Local Threshold + Ink Stroke Recovery (Keeps thin Punjabi matras unbroken while wiping watermark) function cleanDocumentCanvas(ctx, w, h, cutoff, sharpenText, region) { const imgData = ctx.getImageData(0, 0, w, h); const d = imgData.data; const x0 = region ? Math.max(0, region.x) : 0; const y0 = region ? Math.max(0, region.y) : 0; const x1 = region ? Math.min(w - 1, region.x + region.w) : w - 1; const y1 = region ? Math.min(h - 1, region.y + region.h) : h - 1; // Step 1: Build grayscale luminance buffer & integral image for fast local background detection const lumBuf = new Float32Array(w * h); const integral = new Float64Array((w + 1) * (h + 1)); for (let y = 0; y < h; y++) { let rowSum = 0; for (let x = 0; x < w; x++) { const i = (y * w + x) * 4; const l = 0.299 * d[i] + 0.587 * d[i + 1] + 0.114 * d[i + 2]; lumBuf[y * w + x] = l; rowSum += l; integral[(y + 1) * (w + 1) + (x + 1)] = integral[y * (w + 1) + (x + 1)] + rowSum; } } // Radius 14 (~29x29 window): larger than font stroke thickness, smaller than watermark band const r = 14; const contrastDelta = Math.max(12, Math.round((220 - cutoff) * 0.45)); for (let y = y0; y <= y1; y++) { const ya = Math.max(0, y - r); const yb = Math.min(h - 1, y + r); for (let x = x0; x <= x1; x++) { const xa = Math.max(0, x - r); const xb = Math.min(w - 1, x + r); const area = (xb - xa + 1) * (yb - ya + 1); const sum = integral[(yb + 1) * (w + 1) + (xb + 1)] - integral[ya * (w + 1) + (xb + 1)] - integral[(yb + 1) * (w + 1) + xa] + integral[ya * (w + 1) + xa]; const localMean = sum / area; const curLum = lumBuf[y * w + x]; const idx = (y * w + x) * 4; // If pixel is significantly darker than its immediate surroundings, it is real text ink if (curLum < localMean - contrastDelta || curLum < 115) { if (sharpenText) { const ink = Math.max(0, Math.min(255, Math.round((curLum / Math.max(1, localMean)) * 95))); d[idx] = ink; d[idx + 1] = ink; d[idx + 2] = ink; } } else { // Broad watermark band or paper background -> pure white d[idx] = 255; d[idx + 1] = 255; d[idx + 2] = 255; } } } ctx.putImageData(imgData, 0, 0); }
